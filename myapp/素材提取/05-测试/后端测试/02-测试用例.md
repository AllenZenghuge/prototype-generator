### TC-BE-09: Pydantic 模型 — ParseResponse 反序列化（成功）

```python
def test_parse_response_deserialize_success():
    json_str = '{"success":true,"data":{"platform":"twitter","author":{"name":"Test","screen_name":"test"},"text":"Hello","videos":[],"images":[],"source_url":"https://x.com/test/status/1"}}'
    resp = ParseResponse.model_validate_json(json_str)
```

### TC-BE-12: RateLimiter — 正常间隔通过

```python
@pytest.mark.asyncio
async def test_rate_limiter_normal_flow():
    limiter = RateLimiter(min_interval=0.1)
    wait = await limiter.acquire()
    assert wait == 0.0
```

### TC-BE-13: RateLimiter — 间隔不足等待

```python
@pytest.mark.asyncio
async def test_rate_limiter_waits():
    limiter = RateLimiter(min_interval=0.2)
    await limiter.acquire()
    start = time.monotonic()
    await limiter.acquire()
    elapsed = time.monotonic() - start
    assert elapsed >= 0.2
```

### TC-BE-14: RateLimiter — 突发限制触发冷却

```python
@pytest.mark.asyncio
async def test_rate_limiter_burst():
    limiter = RateLimiter(min_interval=0.01, burst_limit=3, cooldown=0.5)
    for _ in range(3):
        await limiter.acquire()
    with pytest.raises(RateLimitError):
        await limiter.acquire()
```
