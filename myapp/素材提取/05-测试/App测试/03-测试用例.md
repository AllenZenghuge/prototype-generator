### TC-APP-01: ExtractViewModel — 初始状态

```swift
func test_extractViewModel_initialState() {
    let vm = ExtractViewModel()
    XCTAssertEqual(vm.inputURL, "")
    XCTAssertEqual(vm.selectedTab, .video)
    XCTAssertFalse(vm.saveCompleted)
    // state 应为 idle
    if case .idle = vm.state { } else { XCTFail("Expected idle state") }
}
```

### TC-APP-02: ExtractViewModel — 空 URL 解析

```swift
func test_extractViewModel_parseEmptyURL() async {
    let vm = ExtractViewModel()
    vm.inputURL = ""
    await vm.parse()
    // 应转为 error 状态
    if case .error(let msg) = vm.state {
        XCTAssertTrue(msg.contains("请输入链接地址"))
    } else {
        XCTFail("Expected error state")
    }
}
```

### TC-APP-04: ExtractViewModel — reset 恢复初始状态

### TC-APP-05: ParseResponse — 成功 JSON 反序列化

```swift
func test_parseResponse_decodeSuccess() throws {
    let json = """
    {"success":true,"data":{"platform":"twitter","author":{"name":"Test","screen_name":"test"},"text":"Hello world","videos":[],"images":[],"source_url":"https://x.com/a/status/1"}}
    """
    let data = json.data(using: .utf8)!
    let response = try JSONDecoder().decode(ParseResponse.self, from: data)
    XCTAssertTrue(response.success)
    XCTAssertEqual(response.data?.text, "Hello world")
    XCTAssertEqual(response.data?.author?.screenName, "test")
}
```

### TC-APP-06: ParseResponse — 失败 JSON 反序列化

```swift
func test_parseResponse_decodeFailure() throws {
    let json = """
    {"success":false,"error":"invalid_url","message":"请输入有效的 X 推文链接"}
    """
    let data = json.data(using: .utf8)!
    let response = try JSONDecoder().decode(ParseResponse.self, from: data)
    XCTAssertFalse(response.success)
    XCTAssertEqual(response.error, "invalid_url")
    XCTAssertNil(response.data)
}
```
